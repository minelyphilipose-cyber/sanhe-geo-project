 147:     private void runMockFlow(Long versionId) {
 148:         log.info("Presale mock generate start, versionId={}, delay={}ms", versionId, mockDelayMs);
 149:         markRunningForMock(versionId);
 150: 
 151:         try {
 152:             Thread.sleep(mockDelayMs);
 153:         } catch (InterruptedException e) {
 154:             Thread.currentThread().interrupt();
 155:             markFailed(versionId, FAILURE_CATEGORY_UNEXPECTED_ERROR, truncateReason("Generation interrupted"));
 156:             return;
 157:         }
 158: 
 159:         String rawJson;
 160:         String computedJson;
 161:         String editableJson;
 162:         try {
 163:             FixturePayload payload = loadFixturePayload();
 164:             rawJson = payload.rawJson();
 165:             computedJson = computedSnapshotEnricher.enrichAndValidate(
 166:                     versionId, rawJson, payload.computedJson(), resolveAllowSyntheticFallback()
 167:             );
 168:             editableJson = l3InitService.derive(rawJson, computedJson);
 169:         } catch (Exception e) {
 170:             log.error("Failed to build presale snapshot, fixturePath={}", mockFixturePath, e);
 171:             markFailed(versionId, FAILURE_CATEGORY_SNAPSHOT_BUILD_ERROR,
 172:                     truncateReason("Snapshot build failed: " + e.getMessage()));
 173:             return;
 174:         }
 175: 
 176:         PresaleReportVersion current = versionMapper.selectById(versionId);
 177:         int totalCalls = current == null || current.getTotalLlmCalls() == null
 178:                 ? 0 : current.getTotalLlmCalls();
 179: 
 180:         PresaleReportVersion update = new PresaleReportVersion();
 181:         update.setId(versionId);
 182:         update.setGenerationStatus(PresaleGenerateStatus.DONE.name());
 183:         update.setGenerationStage(null);
 184:         update.setCompletedLlmCalls(totalCalls);
 185:         update.setTotalLlmCalls(totalCalls);
 186:         update.setBatch1CompletedCalls(current == null ? null : current.getBatch1TotalCalls());
 187:         update.setBatch2CompletedCalls(current == null ? null : current.getBatch2TotalCalls());
 188:         update.setIsDegraded(false);
 189:         update.setRawSnapshotJson(rawJson);
 190:         update.setComputedSnapshotJson(computedJson);
 191:         update.setEditableContentJson(editableJson);
 192:         update.setUpdatedAt(LocalDateTime.now());
 193:         versionMapper.updateById(update);
 194: 
 195:         log.info("Presale mock generate done, versionId={}", versionId);
 196:     }

 198:     private void runRealSkeletonFlow(Long versionId, Long operatorUserId, boolean isManager) {
 199:         PreflightResult preflight = preflight(versionId);
 200:         if (!preflight.success()) {
 201:             markFailed(versionId, FAILURE_CATEGORY_CONFIG_MISSING,
 202:                     truncateReason("CONFIG_MISSING: " + preflight.failureReason()));
 203:             return;
 204:         }
 205: 
 206:         markRunning(versionId, preflight.totalUpperBoundCalls(), preflight.batch1TotalCalls());
 207:         PresaleReportVersion version = versionMapper.selectById(versionId);
 208:         PresaleReport report = version == null ? null : reportMapper.selectById(version.getReportId());
 209:         if (version == null || report == null) {
 210:             markFailed(versionId, FAILURE_CATEGORY_CONFIG_MISSING, "CONFIG_MISSING: report/version not found during batch1");
 211:             return;
 212:         }
 213:         Batch1ExecutionResult batch1Result = executeBatch1(version, report, operatorUserId, isManager, preflight);
 214:         if (batch1Result.stopPipeline) {
 215:             return;
 216:         }
 217: 
 218:         enterStage(versionId, STAGE_COMPETITOR_EXTRACT, "skeleton only");
 219: 
 220:         List<String> extractedCompetitors = extractTopCompetitorsFromBatch1(versionId, report.getBrandName());
 221:         int extractedCompetitorCount = extractedCompetitors.size();
 222:         int batch2TotalCalls = preflight.platformCount() * preflight.competitorPromptCount() * extractedCompetitorCount * 2;
 223:         updateAfterCompetitorExtract(
 224:                 versionId,
 225:                 extractedCompetitorCount,
 226:                 batch2TotalCalls,
 227:                 preflight.batch1TotalCalls() + batch2TotalCalls
 228:         );
 229: 
 230:         if (extractedCompetitorCount > 0) {
 231:             Batch2ExecutionResult batch2Result = executeBatch2(
 232:                     versionId,
 233:                     report,
 234:                     operatorUserId,
 235:                     isManager,
 236:                     extractedCompetitors,
 237:                     preflight.competitorPromptCount()
 238:             );
 239:             if (batch2Result.stopPipeline) {
 240:                 return;
 241:             }
 242:         } else {
 243:             markCompetitorExtractEmpty(versionId);
 244:             log.info("PR-3 stage C skeleton skip batch2 because extracted competitors is 0, versionId={}", versionId);
 245:         }
 246: 
 247:         enterStage(versionId, STAGE_L1_AGGREGATE, "skeleton only");
 248:         enterStage(versionId, STAGE_L2_COMPUTE, "skeleton only");
 249:         enterStage(versionId, STAGE_L3_INIT, "skeleton only");
 250: 
 251:         markFailed(versionId, FAILURE_CATEGORY_STAGE_D_CHECKPOINT,
 252:                 "PR-3 stage D checkpoint: batch2/l1/l2/l3 not implemented yet");
 253:         log.info("PR-3 stage C skeleton finished with placeholder failure, versionId={}, operatorUserId={}, isManager={}",
 254:                 versionId, operatorUserId, isManager);
 255:     }

 257:     private PreflightResult preflight(Long versionId) {
 258:         PresaleReportVersion version = versionMapper.selectById(versionId);
 259:         if (version == null) {
 260:             return PreflightResult.fail("version not found: " + versionId);
 261:         }
 262:         PresaleReport report = reportMapper.selectById(version.getReportId());
 263:         if (report == null) {
 264:             return PreflightResult.fail("report not found: " + version.getReportId());
 265:         }
 266:         if (report.getBrandName() == null || report.getBrandName().isBlank()) {
 267:             return PreflightResult.fail("report.brand_name is blank");
 268:         }
 269: 
 270:         int platformCount = countEnabledPlatforms();
 271:         if (platformCount < 1) {
 272:             return PreflightResult.fail("enabled platform count is 0");
 273:         }
 274: 
 275:         int genericPromptCount = countPromptTemplates(0);
 276:         if (genericPromptCount < 1) {
 277:             return PreflightResult.fail("generic prompt count is 0");
 278:         }
 279: 
 280:         int competitorPromptCount = countPromptTemplates(1);
 281:         int batch1TotalCalls = platformCount * genericPromptCount * 2;
 282:         int totalUpperBoundCalls = batch1TotalCalls + (platformCount * competitorPromptCount * 3 * 2);
 283:         return PreflightResult.success(platformCount, competitorPromptCount, batch1TotalCalls, totalUpperBoundCalls);
 284:     }

 286:     private Batch1ExecutionResult executeBatch1(PresaleReportVersion version,
 287:                                                 PresaleReport report,
 288:                                                 Long operatorUserId,
 289:                                                 boolean isManager,
 290:                                                 PreflightResult preflight) {
 291:         Long versionId = version.getId();
 292:         enterStage(versionId, STAGE_BATCH1, "batch1 executing");
 293: 
 294:         List<AiPlatformConfig> platforms = aiPlatformConfigMapper.selectList(
 295:                 new LambdaQueryWrapper<AiPlatformConfig>()
 296:                         .eq(AiPlatformConfig::getEnabled, true)
 297:                         .orderByAsc(AiPlatformConfig::getPlatformCode)
 298:         );
 299:         List<PresalePromptTemplate> templates = promptTemplateMapper.selectList(
 300:                 new LambdaQueryWrapper<PresalePromptTemplate>()
 301:                         .eq(PresalePromptTemplate::getEnabled, 1)
 302:                         .eq(PresalePromptTemplate::getHasCompetitorVar, 0)
 303:                         .orderByAsc(PresalePromptTemplate::getSortOrder)
 304:                         .orderByAsc(PresalePromptTemplate::getId)
 305:         );
 306: 
 307:         int qGen = templates.size();
 308:         int batch1CompletedCalls = 0;
 309:         int batch2CompletedCalls = 0;
 310:         Set<String> degradedPlatforms = new LinkedHashSet<>();
 311:         List<PlatformBatchState> platformStates = new ArrayList<>();
 312: 
 313:         for (AiPlatformConfig platform : platforms) {
 314:             PlatformBatchState state = new PlatformBatchState(platform.getPlatformCode(), qGen);
 315:             platformStates.add(state);
 316: 
 317:             for (PresalePromptTemplate template : templates) {
 318:                 if (state.degraded) {
 319:                     insertSkippedCall(versionId, 1, platform.getPlatformCode(), template.getId(), "",
 320:                             "QUERY");
 321:                     insertSkippedCall(versionId, 1, platform.getPlatformCode(), template.getId(), "",
 322:                             "ANALYZE");
 323:                     batch1CompletedCalls += 2;
 324:                     updateBatchProgress(versionId, batch1CompletedCalls, batch2CompletedCalls, degradedPlatforms, false);
 325:                     continue;
 326:                 }
 327: 
 328:                 String renderedPrompt = promptTemplateRenderer.render(
 329:                         template.getPromptContent(),
 330:                         template.getPromptCode(),
 331:                         1,
 332:                         report.getBrandName(),
 333:                         report.getIndustry(),
 334:                         report.getIndustryRole(),
 335:                         report.getRegion(),
 336:                         ""
 337:                 );
 338: 
 339:                 PlatformCallContext ctx = new PlatformCallContext(
 340:                         versionId,
 341:                         1,
 342:                         platform.getPlatformCode(),
 343:                         template.getId(),
 344:                         "",
 345:                         report.getBrandName(),
 346:                         operatorUserId,
 347:                         isManager
 348:                 );
 349: 
 350:                 PresaleAiCall queryCall;
 351:                 LlmCallResult queryResult;
 352:                 try {
 353:                     queryResult = llmInvoker.query(ctx, renderedPrompt);
 354:                     queryCall = insertCall(
 355:                             versionId, 1, platform.getPlatformCode(), template.getId(), "",
 356:                             "QUERY", null, queryResult, null
 357:                     );
 358:                 } catch (LlmInvokeException ex) {
 359:                     insertFailedCall(versionId, 1, platform.getPlatformCode(), template.getId(), "",
 360:                             "QUERY", null, ex.getMessage());
 361:                     state.processedPrompts++;
 362:                     state.failedPrompts++;
 363:                     batch1CompletedCalls += 1;
 364:                     if (shouldDegrade(state)) {
 365:                         state.degraded = true;
 366:                         degradedPlatforms.add(state.platformCode);
 367:                     }
 368:                     updateBatchProgress(versionId, batch1CompletedCalls, batch2CompletedCalls, degradedPlatforms, false);
 369:                     if (degradedPlatforms.size() >= 4) {
 370:                         updateBatchProgress(versionId, batch1CompletedCalls, batch2CompletedCalls, degradedPlatforms, true);
 371:                         markTooManyDegradedFailed(versionId, degradedPlatforms);
 372:                         return Batch1ExecutionResult.stop();
 373:                     }
 374:                     continue;
 375:                 }
 376: 
 377:                 batch1CompletedCalls += 1;
 378:                 updateBatchProgress(versionId, batch1CompletedCalls, batch2CompletedCalls, degradedPlatforms, false);
 379: 
 380:                 try {
 381:                     LlmCallResult analyzeResult = llmInvoker.analyze(ctx, renderedPrompt, queryResult.rawResponse());
 382:                     PresaleAiCall analyzeCall = insertCall(
 383:                             versionId, 1, platform.getPlatformCode(), template.getId(), "",
 384:                             "ANALYZE", queryCall.getId(), analyzeResult, null
 385:                     );
 386:                     insertPromptResultSuccess(versionId, 1, platform.getPlatformCode(), template.getId(), "",
 387:                             queryCall.getId(), analyzeCall.getId(), analyzeResult.rawResponse());
 388:                 } catch (LlmInvokeException | AnalyzeParseException ex) {
 389:                     PresaleAiCall analyzeCall = insertFailedCall(versionId, 1, platform.getPlatformCode(), template.getId(), "",
 390:                             "ANALYZE", queryCall.getId(), ex.getMessage());
 391:                     insertPromptResultAnalyzeFailed(versionId, 1, platform.getPlatformCode(), template.getId(), "",
 392:                             queryCall.getId(), analyzeCall.getId());
 393:                     state.failedPrompts++;
 394:                 }
 395: 
 396:                 state.processedPrompts++;
 397:                 batch1CompletedCalls += 1;
 398:                 if (shouldDegrade(state)) {
 399:                     state.degraded = true;
 400:                     degradedPlatforms.add(state.platformCode);
 401:                 }
 402:                 updateBatchProgress(versionId, batch1CompletedCalls, batch2CompletedCalls, degradedPlatforms, false);
 403: 
 404:                 if (degradedPlatforms.size() >= 4) {
 405:                     updateBatchProgress(versionId, batch1CompletedCalls, batch2CompletedCalls, degradedPlatforms, true);
 406:                     markTooManyDegradedFailed(versionId, degradedPlatforms);
 407:                     return Batch1ExecutionResult.stop();
 408:                 }
 409:             }
 410:         }
 411:         updateBatchProgress(versionId, batch1CompletedCalls, batch2CompletedCalls, degradedPlatforms, true);
 412:         return Batch1ExecutionResult.continuePipeline();
 413:     }

 474:                             versionId,
 475:                             2,
 476:                             platform.getPlatformCode(),
 477:                             template.getId(),
 478:                             competitorName,

 480:                             operatorUserId,

 482:                     );

 484:                     PresaleAiCall queryCall;
 485:                     LlmCallResult queryResult;
 486:                     try {
 487:                         queryResult = llmInvoker.query(ctx, renderedPrompt);
 488:                         queryCall = insertCall(
 489:                                 versionId, 2, platform.getPlatformCode(), template.getId(), competitorName,
 490:                                 "QUERY", null, queryResult, null
 491:                         );
 492:                     } catch (LlmInvokeException ex) {
 493:                         insertFailedCall(versionId, 2, platform.getPlatformCode(), template.getId(), competitorName,
 494:                                 "QUERY", null, ex.getMessage());
 495:                         state.processedPrompts++;
 496:                         state.failedPrompts++;
 497:                         batch2CompletedCalls += 1;
 498:                         if (shouldDegrade(state)) {
 499:                             state.degraded = true;
 500:                             batch2DegradedPlatforms.add(state.platformCode);
 501:                             displayDegradedPlatforms.add(state.platformCode);
 502:                         }
 503:                         updateBatchProgress(versionId, batch1CompletedCalls, batch2CompletedCalls, displayDegradedPlatforms, false);
 504:                         if (batch2DegradedPlatforms.size() >= 4) {
 505:                             updateBatchProgress(versionId, batch1CompletedCalls, batch2CompletedCalls, displayDegradedPlatforms, true);
 506:                             markTooManyDegradedFailed(versionId, displayDegradedPlatforms);

 508:                         }
 509:                         continue;
 510:                     }
 511:

 513:                     updateBatchProgress(versionId, batch1CompletedCalls, batch2CompletedCalls, displayDegradedPlatforms, false);
 514: 
 515:                     try {
 516:                         LlmCallResult analyzeResult = llmInvoker.analyze(ctx, renderedPrompt, queryResult.rawResponse());

 518:                                 versionId, 2, platform.getPlatformCode(), template.getId(), competitorName,

 726:     private List<String> extractTopCompetitorsFromBatch1(Long versionId, String brandName) {
 727:         List<PresaleAiPromptResult> rows = aiPromptResultMapper.selectList(
 728:                 new LambdaQueryWrapper<PresaleAiPromptResult>()
 729:                         .eq(PresaleAiPromptResult::getVersionId, versionId)
 730:                         .eq(PresaleAiPromptResult::getBatchNo, 1)
 731:                         .isNotNull(PresaleAiPromptResult::getIsMentioned)
 732:         );
 733:         if (rows == null || rows.isEmpty()) {
 734:             return List.of();
 735:         }
 736: 
 737:         String normalizedBrand = normalizeName(brandName);
 738:         Map<String, Integer> countByNormalized = new HashMap<>();
 739:         Map<String, String> displayByNormalized = new HashMap<>();
 740:         for (PresaleAiPromptResult row : rows) {
 741:             String mentionedCompetitors = row.getMentionedCompetitors();
 742:             if (mentionedCompetitors == null || mentionedCompetitors.isBlank()) {

 748:                 if (!node.isArray()) {
 749:                     continue;
 750:                 }
 751: 
 752:                 Set<String> rowDedup = new HashSet<>();
 753:                 for (JsonNode item : node) {
 754:                     if (!item.isTextual()) {
 755:                         continue;
 756:                     }
 757:                     String display = item.asText() == null ? "" : item.asText().trim();
 758:                     if (display.isEmpty()) {
 759:                         continue;
 760:                     }
 761:                     String normalized = normalizeName(display);
 762:                     if (normalized.isEmpty() || normalized.equals(normalizedBrand)) {
 763:                         continue;
 764:                     }
 765:                     if (!rowDedup.add(normalized)) {
 766:                         continue;
 767:                     }
 768:                     countByNormalized.merge(normalized, 1, Integer::sum);
 769:                     displayByNormalized.putIfAbsent(normalized, display);
 770:                 }
 771:             } catch (Exception ex) {
 772:                 log.warn("Skip invalid mentioned_competitors json, versionId={}, promptResultId={}",
 773:                         versionId, row.getId(), ex);
 774:             }
 775:         }
 776: 
 777:         return countByNormalized.entrySet().stream()
 778:                 .sorted(Comparator
 779:                         .comparing(Map.Entry<String, Integer>::getValue, Comparator.reverseOrder())
 780:                         .thenComparing(Map.Entry::getKey))
 781:                 .limit(3)
 782:                 .map(entry -> displayByNormalized.getOrDefault(entry.getKey(), entry.getKey()))
 783:                 .toList();
 784:     }
